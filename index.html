<script>
(function () {
  const canvas = document.getElementById('cafe-network-canvas');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');

  const NUM_POINTS = 45;
  const MAX_DIST = 140;
  const SPEED = 0.5;
  const points = [];

  const SYMBOLS = ['μ', 'σ²', 'var(x)', 'P(A)', 'H₀·H₁', 'x̄', 'R', 'n'];
  const NUM_SYMBOLS = 12;
  const symbols = [];

  const mouse = { x: null, y: null, radius: 150 };

  function resize() {
    canvas.width = canvas.offsetWidth * devicePixelRatio;
    canvas.height = canvas.offsetHeight * devicePixelRatio;
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.scale(devicePixelRatio, devicePixelRatio);
  }

  function initPoints() {
    points.length = 0;
    const w = canvas.offsetWidth;
    const h = canvas.offsetHeight;
    for (let i = 0; i < NUM_POINTS; i++) {
      points.push({
        x: Math.random() * w,
        y: Math.random() * h,
        vx: (Math.random() - 0.5) * SPEED,
        vy: (Math.random() - 0.5) * SPEED
      });
    }
  }

  function initSymbols() {
    symbols.length = 0;
    const w = canvas.offsetWidth;
    const h = canvas.offsetHeight;
    for (let i = 0; i < NUM_SYMBOLS; i++) {
      symbols.push({
        x: Math.random() * w,
        y: Math.random() * h,
        vx: (Math.random() - 0.5) * SPEED * 0.5,
        vy: (Math.random() - 0.5) * SPEED * 0.5,
        char: SYMBOLS[Math.floor(Math.random() * SYMBOLS.length)],
        size: 14 + Math.random() * 14,
        opacity: 0.14 + Math.random() * 0.16
      });
    }
  }

  canvas.addEventListener('mousemove', (e) => {
    const rect = canvas.getBoundingClientRect();
    mouse.x = e.clientX - rect.left;
    mouse.y = e.clientY - rect.top;
  });
  canvas.addEventListener('mouseleave', () => { mouse.x = null; mouse.y = null; });

  function tick() {
    const w = canvas.offsetWidth;
    const h = canvas.offsetHeight;
    ctx.clearRect(0, 0, w, h);

    for (const p of points) {
      p.x += p.vx;
      p.y += p.vy;
      if (p.x < 0 || p.x > w) p.vx *= -1;
      if (p.y < 0 || p.y > h) p.vy *= -1;

      if (mouse.x !== null) {
        const dx = mouse.x - p.x;
        const dy = mouse.y - p.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < mouse.radius) {
          p.x += dx * 0.0015;
          p.y += dy * 0.0015;
        }
      }
    }

    for (let i = 0; i < points.length; i++) {
      for (let j = i + 1; j < points.length; j++) {
        const a = points[i], b = points[j];
        const dx = a.x - b.x;
        const dy = a.y - b.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < MAX_DIST) {
          const t = dist / MAX_DIST;
          const opacity = (1 - t) * 0.28;
          ctx.strokeStyle = `rgba(107,79,79,${opacity})`;
          ctx.lineWidth = 0.8;
          ctx.beginPath();
          ctx.moveTo(a.x, a.y);
          ctx.lineTo(b.x, b.y);
          ctx.stroke();
        }
      }
    }

    for (const p of points) {
      ctx.beginPath();
      ctx.arc(p.x, p.y, 2, 0, Math.PI * 2);
      ctx.fillStyle = 'rgba(107,79,79,0.5)';
      ctx.fill();
    }

    for (const s of symbols) {
      s.x += s.vx;
      s.y += s.vy;
      if (s.x < -30 || s.x > w + 30) s.vx *= -1;
      if (s.y < -30 || s.y > h + 30) s.vy *= -1;

      ctx.font = `600 ${s.size}px 'Poppins', sans-serif`;
      ctx.fillStyle = `rgba(75,46,43,${s.opacity})`;
      ctx.fillText(s.char, s.x, s.y);
    }

    requestAnimationFrame(tick);
  }

  window.addEventListener('resize', () => {
    resize();
    initPoints();
    initSymbols();
  });

  resize();
  initPoints();
  initSymbols();
  tick();
})();
</script>
